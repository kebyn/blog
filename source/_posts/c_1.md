### 功能

输入文件名后创建c文件，并用vc6打开c文件

<!-- more -->

```
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <windows.h>
int main ()
{
        FILE*fp;
        char ch ,filename[256];
        printf("filename:");
        scanf("%s",filename);
        strcat(filename,".c");
        if ((fp=fopen(filename,"w")) == NULL)
        {
                printf ("erro\n");
                return 0;
                exit (0);
        }
        ch=getchar();
        printf("EOF for #\n");
        ch=getchar();
        if(ch != '\n')
        {
        while(ch != '#')
        {
                fputc(ch,fp);
//    			putchar(ch);
                ch=getchar();
        }
        }
        fclose(fp);
        putchar(256);
		ShellExecute(NULL,"open",filename,"D:/VC98/Bin/MSDEV.EXE",NULL,SW_SHOWNORMAL);
 return 0;

 }
```
